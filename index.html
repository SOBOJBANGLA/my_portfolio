<?php

namespace App\Http\Controllers;

use App\Models\Employee;
use App\Models\Payroll;
use Illuminate\Http\Request;

class PayrollController extends Controller
{
    /**
     * Process Monthly Salary Calculation for Wholesale HRMS (Laravel Architecture)
     */
    public function calculateMonthlyPayroll(Request $request)
    {
        $validated = $request->validate([
            'month' => 'required|date_format:Y-m',
            'department_id' => 'nullable|exists:departments,id',
        ]);

        $employees = Employee::with(['attendances', 'bonuses', 'deductions'])
            ->where('status', 'active')
            ->get();

        $processedPayrolls = [];
        foreach ($employees as $employee) {
            $netSalary = $employee->base_salary + $employee->total_bonus - $employee->total_deductions;
            $processedPayrolls[] = Payroll::updateOrCreate(
                ['employee_id' => $employee->id, 'payroll_month' => $validated['month']],
                ['net_salary' => $netSalary, 'status' => 'generated']
            );
        }

        return response()->json([
            'success' => true,
            'message' => 'Payroll generated successfully',
            'data' => $processedPayrolls
        ]);
    }
}

import { NextRequest, NextResponse } from 'next/server';

interface BidRequest {
  publisherId: string;
  adSlot: string;
  deviceType: 'mobile' | 'desktop' | 'dooh';
  floorPrice: number;
}

export async function POST(req: NextRequest) {
  try {
    const body: BidRequest = await req.json();
    
    // High-speed DSP auction matching engine with sub-millisecond response
    const winningBid = await executeRtbAuction({
      slot: body.adSlot,
      device: body.deviceType,
      floor: body.floorPrice
    });

    return NextResponse.json({
      success: true,
      creativeUrl: winningBid.creativeUrl,
      trackingId: winningBid.impressionBeacon,
      cpm: winningBid.cpmRate
    }, { status: 200 });
  } catch (error) {
    return NextResponse.json({ error: 'Auction dispatch failed' }, { status: 500 });
  }
}
-- Enterprise Architecture: Multi-Tenant HRMS, ISP & Wholesale POS Schema

CREATE TABLE `tenants` (
  `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  `company_name` VARCHAR(191) NOT NULL,
  `domain` VARCHAR(191) UNIQUE NOT NULL,
  `status` ENUM('active', 'suspended', 'trial') DEFAULT 'active',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `sales_invoices` (
  `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  `tenant_id` BIGINT UNSIGNED NOT NULL,
  `invoice_no` VARCHAR(64) UNIQUE NOT NULL,
  `customer_id` BIGINT UNSIGNED NULL,
  `sub_total` DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  `tax_amount` DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  `discount_amount` DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  `grand_total` DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  `payment_status` ENUM('paid', 'partial', 'due') DEFAULT 'paid',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`tenant_id`) REFERENCES `tenants`(`id`) ON DELETE CASCADE,
  INDEX `idx_tenant_invoice` (`tenant_id`, `invoice_no`),
  INDEX `idx_payment_status` (`payment_status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `invoice_items` (
  `id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  `invoice_id` BIGINT UNSIGNED NOT NULL,
  `product_id` BIGINT UNSIGNED NOT NULL,
  `unit_price` DECIMAL(12,2) NOT NULL,
  `quantity` INT NOT NULL DEFAULT 1,
  `total_price` DECIMAL(12,2) NOT NULL,
  FOREIGN KEY (`invoice_id`) REFERENCES `sales_invoices`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
